import Link from '@docusaurus/Link';
import useBaseUrl from '@docusaurus/useBaseUrl';
import libraryDescription from '@site/src/data/libraryDescription.json';

export const routeForLibrary = (id) => {
  switch (id) {
    case 'core':
      return '/docs/core/dispatcher';
    case 'i18n':
      return '/docs/i18n/$L';
    case 'spotlight':
      return '/docs/spotlight/';
    case 'ui':
      return '/docs/ui/Button';
    case 'webos':
      return '/docs/webos/application';
    case 'sandstone':
      return '/docs/sandstone/ActionGuide';
    case 'moonstone':
      return '/docs/moonstone/BodyText';
    case 'agate':
      return '/docs/agate/ActionGuide';
    case 'limestone':
      return '/docs/limestone/ActionGuide';
    default:
      return `/docs/${id}/`;
  }
};

export const libraryCards = Object.keys(libraryDescription)
  .filter((id) => id !== 'cli' && id !== 'eslint-config-enact')
  .map((id) => ({
    id,
    title: `${id} Library`,
    to: routeForLibrary(id),
  }));

const ApiLibraryCard = ({card}) => {
  const img = useBaseUrl(`/img/package-${card.id}.svg`);
  return (
    <Link className="api-library-card" to={card.to} key={card.id}>
      <img
        className="api-library-icon"
        src={img}
        alt={`${card.id} library`}
      />
      <div className="api-library-body">
        <h3>{card.title}</h3>
        <p className="api-library-version">
          {libraryDescription[card.id].version}
        </p>
        <p className="api-library-description">
          {libraryDescription[card.id].description}
        </p>
      </div>
    </Link>
  );
};
